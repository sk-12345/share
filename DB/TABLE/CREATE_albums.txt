USE share;

-- 既にあれば削除
DROP TABLE IF EXISTS albums;

create table share.`albums` (
  id int(10) unsigned auto_increment not null comment 'アルバムID'
  , title varchar(255) not null comment 'タイトル'
  , description text not null comment '説明'
  , folder_key varchar(255) not null comment 'アルバム鍵'
  , created_by int(10) unsigned not null comment '作成者'
  , created_at datetime default current_timestamp() not null comment '作成時間'
  , updated_at datetime on update CURRENT_TIMESTAMP default current_timestamp() not null comment '更新時間'
  , primary key (id)
)
  comment='アルバム'
  engine InnoDB
  row_format DYNAMIC
  auto_increment 6
  collate utf8mb4_general_ci

-- 既にあれば削除
DROP TABLE IF EXISTS album_photos;

create table share.`album_photos` (
  id int(10) unsigned auto_increment not null comment '写真ID'
  , album_id int(10) unsigned not null comment 'アルバムID'
  , file_name varchar(255) not null comment 'ファイル名前'
  , original_name varchar(255) not null comment '写真名前'
  , created_at datetime default current_timestamp() not null comment '作成時間'
  , primary key (id)
)
  comment='写真'
  engine InnoDB
  row_format DYNAMIC
  auto_increment 7
  collate utf8mb4_general_ci
